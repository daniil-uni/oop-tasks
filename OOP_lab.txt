class Device:
    def __init__(self, name, battery_life):
        self.__name = name
        self.__battery_life = battery_life

    def get_name(self):
        return self.__name

    def get_battery_life(self):
        return self.__battery_life

    def set_name(self, name):
        self.__name = name

    def set_battery_life(self, battery_life):
        self.__battery_life = battery_life


class Smartphone(Device):
    def call(self):
        print("Телефон звонит")

    def use(self):
        if self.get_battery_life() >= 10:
            self.set_battery_life(self.get_battery_life() - 10)
            print(f"Телефон используется, оставшийся заряд: {self.get_battery_life()}%")
        else:
            print("Недостаточно заряда")


class Laptop(Device):
    def compile_code(self):
        print("Ноутбук компилирует код")

    def use(self):
        if self.get_battery_life() >= 10:
            self.set_battery_life(self.get_battery_life() - 10)
            print(f"Ноутбук используется, оставшийся заряд: {self.get_battery_life()}%")
        else:
            print("Недостаточно заряда")


class Tablet(Device):
    def draw(self):
        print("Планшет используется для рисования")

    def use(self):
        if self.get_battery_life() >= 10:
            self.set_battery_life(self.get_battery_life() - 10)
            print(f"Планшет используется, оставшийся заряд: {self.get_battery_life()}%")
        else:
            print("Недостаточно заряда")


devices = (Smartphone("Samsung", 8), Laptop("HP", 50), Tablet("Apple", 30))

devices[0].call()
devices[1].compile_code()
devices[2].draw()

for device in devices:
    device.use()
