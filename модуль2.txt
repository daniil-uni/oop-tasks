class LearningPlatform:
    def __init__(self, name):
        self.name = name
        self.courses = []

    def add_course(self, course):
        self.courses.append(course)


class Course:
    def __init__(self, name, duration):
        self.name = name
        self.duration = duration


course_1 = Course("math", "12 month")
course_2 = Course("engl", "7 month")

learning_platform = LearningPlatform("samga")
learning_platform.add_course(course_1)
learning_platform.add_course(course_2)

class Employee:
    def __init__(self, experience, name):
        self.__experience = experience
        self._name = name

    def get_experience(self):
        return self.__experience

    def display_info(self):
        print(f"Имя: {self._name}, опыт: {self.__experience}")

    def calculate_workload(self, workload):
        raise NotImplementedError("Метод должен быть реализован в подклассе")


class Teacher(Employee):
    def calculate_bonus(self, bonus):
        self.bonus = bonus * 0.1
        return self.bonus

    def calculate_workload(self, workload):
        self.workload = workload * 4
        return self.workload


class Researcher(Employee):
    def calculate_bonus(self, bonus):
        self.bonus = bonus * 0.15
        return self.bonus

    def calculate_workload(self, workload):
        self.workload = workload / 2
        return self.workload


class Professor(Researcher, Teacher):
    pass


teacher1 = Teacher(4, "anna")
researcher1 = Researcher(3, "david")
professor1 = Professor(7, "nura")

teacher1.display_info()
researcher1.display_info()
professor1.display_info()
