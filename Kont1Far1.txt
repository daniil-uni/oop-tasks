Class Vehicle:
	def __init__(_speed, _color):
		self._speed = _speed
		self._color = _color
		return 0

	def set_speed(self):
		self._speed = _speed
		return(self)

	def set_color(self):
		self._color = _color
		return(self)

	def get_speed(self):
		print(self._speed)
		return 0

	def get_color(self):
		print(self._color)
		return 0

Class Car(Vehicle):
	def __init__(_speed, _color):
		self._speed = _speed
		self._color = _color
		return 0

	def drive():
		print("Машина едет")
		return 0

	def move():
		print("Машина двигается")
		return 0
