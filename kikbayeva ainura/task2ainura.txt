class Book:
	def __init__(self, name, author):
		self.name = name
		self.author = author

	def display_info(self):
		print(f"Название: {self.title}, автор: {self.author}")

class Library:
	def __init__(self):
		self.books = []

	def lend_book(self, book):
		print(f"Книга '{book.name}' выдана читателю.")

	def add_book(self, book):
		self.books.append(book)
		print(f"Книга '{book.name}' добавлена в библиотеку.")

	def show_books(self):
		print("Книги в библиотеке:")
		for book in self.books:
			book.display_info()



book1 = Book("Война и мир", "Л.Н.Толстой")
book2 = Book("Мастер и Маргарита", "М.Булгаков")
library = Library()
library.add_book(book1)
library.add_book(book2)

library.show_books()

library.lend_book(book1)

library.show_books()
