	speed = 80
	def deliver(self, cargo : str) -> str:
		return f'Грузовик везёт "{cargo}" по дороге'

class Ship:
	speed = 30
	def deliver(self, cargo : str) -> str:
		return f'Корабль везёт "{cargo}" по морю'

class Plane:
	speed = 800
	def deliver(self, cargo : str) -> str:
		return f'Самолет везёт "{cargo}" по воздуху'

class Logistics:
	def create_transport(self):
		raise NotImplentedError

	def plan_delivery(self, cargo : str, distance_km : int) -> str:
		transport = self.create_transport()
		hours = distance_km / transport.speed
		return f'{transport.deliver(cargo), в пути {hours :. 1f} ч'

class RoadLogistics(Logistics):
	def create_transport(self):
